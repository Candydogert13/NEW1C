структура ЗаписьИстории
    пер Выражение: Строка
    пер Результат: Число
;

метод Скрипт()
    пер ИсторияВычислений: Массив<ЗаписьИстории> = []

    ВыполнитьВычисление("15 + 10", ИсторияВычислений)
    ВыполнитьВычисление("(15 + 10) * 2", ИсторияВычислений)
    ВыполнитьВычисление("100 / 4", ИсторияВычислений)

    ВывестиИсторию(ИсторияВычислений)
;

метод ВыполнитьВычисление(Выражение: Строка, История: Массив<ЗаписьИстории>)
    пер Результат: Число = 0

    если Выражение == "15 + 10"
        Результат = 15 + 10
    иначе если Выражение == "(15 + 10) * 2"
        Результат = (15 + 10) * 2
    иначе если Выражение == "100 / 4"
        Результат = 100 / 4
    иначе
        возврат
    ;

    пер НоваяЗапись = новый ЗаписьИстории(Выражение, Результат)
    История.Добавить(НоваяЗапись)
    Консоль.Записать("%Выражение = %Результат")
;

метод ВывестиИсторию(История: Массив<ЗаписьИстории>)
    Консоль.Записать("--- История вычислений ---")
    для Запись из История
        Консоль.Записать("%Запись.Выражение = %Запись.Результат")
    ;
;
